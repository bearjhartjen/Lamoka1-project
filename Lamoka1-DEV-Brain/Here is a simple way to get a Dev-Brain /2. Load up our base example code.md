## How to set up a workspace 

 1. Download Arudino IDE or similar 

 2. Install FastLED library

 3. Communicate to the board via UF2 

## What next?

 1. Now we need to enable BOOTSEL mode on the Dev-Brain to communicate with your computer, so hold the button on your board then connect the board to your computer. Your computer should then recognize the board as a storage device. 

 2. With BOOTSEL mode on the board and UF2 as your preffered on the IDE, we made you this C++ code so paste this into your IDE: 

#include <Arduino.h>
#include <FastLED.h>

#define LED_PIN 11
#define NUM_LEDS 10
#define BRIGHTNESS 18

CRGB leds[NUM_LEDS];

uint32_t noiseX = 0;
uint32_t noiseT = 0;

const CRGB DEEP_RED = CRGB(125, 0, 12);
const CRGB RED      = CRGB(210, 8, 20);
const CRGB ORANGE   = CRGB(255, 55, 8);
const CRGB AMBER    = CRGB(255, 125, 18);

CRGB mixColor(
    const CRGB &a,
    const CRGB &b,
    uint8_t amount
)
{
    return CRGB(
        lerp8by8(a.r, b.r, amount),
        lerp8by8(a.g, b.g, amount),
        lerp8by8(a.b, b.b, amount)
    );
}

void setup()
{
    FastLED.addLeds<WS2812B, LED_PIN, GRB>(
        leds,
        NUM_LEDS
    );

    FastLED.setBrightness(BRIGHTNESS);
    FastLED.clear(true);

    noiseX = 123456;
    noiseT = 654321;
}

void loop()
{
    EVERY_N_MILLISECONDS(20)
    {
        noiseX += 100;
        noiseT += 50;

        for (uint8_t i = 0; i < NUM_LEDS; i++)
        {
            // =================================================
            // LARGE FIELD
            // =================================================

            uint16_t largeNoise =
                inoise16(
                    noiseX + (uint32_t)i * 17000UL,
                    noiseT
                );

            uint8_t large =
                largeNoise >> 8;


            // =================================================
            // MEDIUM FIELD
            // =================================================

            uint16_t mediumNoise =
                inoise16(
                    noiseX * 2 + (uint32_t)i * 8500UL,
                    noiseT * 2
                );

            uint8_t medium =
                mediumNoise >> 8;


            // =================================================
            // GENTLE COMBINATION
            // =================================================

            uint8_t field =
                scale8(large, 215) +
                scale8(medium, 40);


            // =================================================
            // COMPRESS COLOR RANGE
            // =================================================

            // Instead of allowing the noise to use almost
            // the entire palette, keep it centered.
            //
            // Result:
            // rich movement without isolated color extremes.

            field =
                70 +
                scale8(
                    field,
                    125
                );


            // =================================================
            // COLOR
            // =================================================

            CRGB color;

            if (field < 115)
            {
                uint8_t amount =
                    map(
                        field,
                        70,
                        115,
                        0,
                        255
                    );

                color =
                    mixColor(
                        DEEP_RED,
                        RED,
                        amount
                    );
            }
            else if (field < 160)
            {
                uint8_t amount =
                    map(
                        field,
                        115,
                        160,
                        0,
                        255
                    );

                color =
                    mixColor(
                        RED,
                        ORANGE,
                        amount
                    );
            }
            else
            {
                uint8_t amount =
                    map(
                        field,
                        160,
                        195,
                        0,
                        255
                    );

                color =
                    mixColor(
                        ORANGE,
                        AMBER,
                        amount
                    );
            }


            // =================================================
            // VERY SUBTLE DETAIL
            // =================================================

            uint16_t detailNoise =
                inoise16(
                    noiseX * 3 +
                    (uint32_t)i * 4500UL,
                    noiseT * 3
                );

            uint8_t detail =
                detailNoise >> 8;

            if (detail > 210)
            {
                uint8_t highlight =
                    scale8(
                        detail - 210,
                        8
                    );

                color.r =
                    qadd8(
                        color.r,
                        highlight
                    );

                color.g =
                    qadd8(
                        color.g,
                        highlight / 2
                    );
            }


            // =================================================
            // CONSTANT-ISH BRIGHTNESS
            // =================================================

            color.nscale8(230);

            leds[i] = color;
        }


        // =====================================================
        // MAKE THE STRIP ONE CONTINUOUS LIGHT FIELD
        // =====================================================

        blur1d(
            leds,
            NUM_LEDS,
            255
        );

        FastLED.show();
    }
}
