This will be a good way for you to ensure everything was assembled and fabricated correctly

## How to set up a workspace 

 1. Download Arudino IDE or similar 

 2. Install FastLED library

 3. Communicate to the board via UF2 

## What next?

 1. Now we need to enable BOOTSEL mode on the Dev-Brain to communicate with your computer, so hold the button on your board then connect the board to your computer. Your computer should then recognize the board as a storage device. 

 2. With BOOTSEL mode on the board and UF2 as your preffered on the IDE, we made you this C++ code so paste this into your IDE: 

#include <Arduino.h>
#include <FastLED.h>

/*
 * ================================================================
 *                    LAMOKA1 DEV BRAIN
 *                       FIRE FIELD
 * ================================================================
 *
 *  Hardware:
 *
 *      RP2350
 *      WS2812B x10
 *      DATA -> GPIO 11
 *
 *  This is intentionally designed around a very small LED strip.
 *
 *  The goal is NOT random flickering.
 *
 *  The goal is:
 *
 *      deep red
 *          ↓
 *      red
 *          ↓
 *      orange
 *          ↓
 *      amber
 *
 *  with the entire strip behaving like one continuous field.
 *
 * ================================================================
 */


/* ================================================================
 *                         HARDWARE
 * ================================================================ */

#define LED_PIN       11
#define NUM_LEDS      10

#define BRIGHTNESS    18


/* ================================================================
 *                         ANIMATION
 * ================================================================ */

#define FRAME_MS      24


/* ================================================================
 *                         LED ARRAY
 * ================================================================ */

CRGB leds[NUM_LEDS];


/* ================================================================
 *                         COLOR PALETTE
 * ================================================================ */

const CRGB DEEP_RED = CRGB(95,  0,  7);
const CRGB RED      = CRGB(205, 5,  18);
const CRGB ORANGE   = CRGB(255, 52, 6);
const CRGB AMBER    = CRGB(255, 125, 15);
const CRGB HOT      = CRGB(255, 175, 35);


/* ================================================================
 *                         NOISE STATE
 * ================================================================ */

uint32_t noiseTime = 0;


/*
 * These determine how much the flame moves from LED to LED.
 *
 * Larger spacing:
 *      more independent movement
 *
 * Smaller spacing:
 *      smoother continuous field
 */

const uint32_t NOISE_SPACING = 10500;


/* ================================================================
 *                         COLOR MIX
 * ================================================================ */

static inline CRGB mixColor(
    const CRGB &a,
    const CRGB &b,
    uint8_t amount
)
{
    return CRGB(
        lerp8by8(a.r, b.r, amount),
        lerp8by8(a.g, b.g, amount),
        lerp8by8(a.b, b.b, amount)
    );
}


/* ================================================================
 *                         FAST PALETTE
 * ================================================================ */

static inline CRGB firePalette(
    uint8_t heat
)
{
    /*
     * 0-63:
     *
     * Deep red -> red
     */

    if (heat < 64)
    {
        return mixColor(
            DEEP_RED,
            RED,
            heat << 2
        );
    }


    /*
     * 64-139:
     *
     * Red -> orange
     */

    if (heat < 140)
    {
        return mixColor(
            RED,
            ORANGE,
            (heat - 64) << 1
        );
    }


    /*
     * 140-209:
     *
     * Orange -> amber
     */

    if (heat < 210)
    {
        return mixColor(
            ORANGE,
            AMBER,
            (heat - 140) * 3
        );
    }


    /*
     * 210-255:
     *
     * Amber -> hot core.
     */

    return mixColor(
        AMBER,
        HOT,
        (heat - 210) * 5
    );
}


/* ================================================================
 *                    NEIGHBOR BLENDING
 * ================================================================ */

static void smoothField()
{
    /*
     * We deliberately do NOT use blur1d().
     *
     * There are only ten pixels.
     *
     * A tiny explicit 3-pixel spatial filter is cheaper,
     * predictable, and gives us control over how much the
     * flame connects together.
     */

    CRGB previous = leds[0];

    for (
        uint8_t i = 1;
        i < NUM_LEDS - 1;
        i++
    )
    {
        CRGB current = leds[i];
        CRGB next = leds[i + 1];


        leds[i] = CRGB(
            (
                previous.r +
                current.r * 2 +
                next.r
            ) >> 2,

            (
                previous.g +
                current.g * 2 +
                next.g
            ) >> 2,

            (
                previous.b +
                current.b * 2 +
                next.b
            ) >> 2
        );


        previous = current;
    }
}


/* ================================================================
 *                    FIRE FIELD GENERATOR
 * ================================================================ */

static void renderFire()
{
    for (
        uint8_t i = 0;
        i < NUM_LEDS;
        i++
    )
    {
        /*
         * --------------------------------------------------------
         * LARGE ORGANIC FIELD
         * --------------------------------------------------------
         *
         * This is the main movement.
         */

        uint16_t n =
            inoise16(
                noiseTime +
                (uint32_t)i *
                NOISE_SPACING
            );


        uint8_t heat =
            n >> 8;


        /*
         * --------------------------------------------------------
         * COMPRESS THE FIELD
         * --------------------------------------------------------
         *
         * Raw Perlin noise has too much range for this effect.
         *
         * Keeping the result mostly between approximately
         * 70 and 210 gives us a rich fire without giant flashes.
         */

        heat =
            65 +
            scale8(
                heat,
                150
            );


        /*
         * --------------------------------------------------------
         * SMALL DETAIL FIELD
         * --------------------------------------------------------
         *
         * Instead of another expensive full noise field,
         * derive a subtle local variation from the main value.
         */

        uint8_t detail =
            sin8(
                (uint8_t)(
                    noiseTime >> 5
                ) +
                i * 19
            );


        /*
         * Only a small amount of detail is allowed.
         */

        heat =
            qadd8(
                heat,
                scale8(
                    detail,
                    18
                )
            );


        /*
         * --------------------------------------------------------
         * NATURAL HOT-SPOT CURVE
         * --------------------------------------------------------
         *
         * Push the upper part of the field slightly harder.
         *
         * This gives the flame occasional amber/hot regions
         * without turning the entire strip yellow.
         */

        if (
            heat > 175
        )
        {
            uint8_t extra =
                heat - 175;


            heat =
                qadd8(
                    heat,
                    extra >> 2
                );
        }


        /*
         * --------------------------------------------------------
         * EDGE BEHAVIOR
         * --------------------------------------------------------
         *
         * The ends of a physical strip tend to look slightly
         * detached. A tiny reduction at the extreme ends makes
         * the whole ten-pixel strip read as one object.
         */

        if (
            i == 0 ||
            i == NUM_LEDS - 1
        )
        {
            heat =
                scale8(
                    heat,
                    225
                );
        }


        /*
         * --------------------------------------------------------
         * COLOR
         * --------------------------------------------------------
         */

        CRGB color =
            firePalette(
                heat
            );


        /*
         * --------------------------------------------------------
         * MICRO FLICKER
         * --------------------------------------------------------
         *
         * Very restrained.
         *
         * This prevents the effect from looking like a smooth
         * gradient while avoiding annoying digital flicker.
         */

        uint8_t flicker =
            sin8(
                (uint8_t)(
                    (noiseTime >> 4) +
                    i * 37
                )
            );


        if (
            flicker > 225
        )
        {
            uint8_t amount =
                (flicker - 225) >> 2;


            color.r =
                qadd8(
                    color.r,
                    amount
                );


            color.g =
                qadd8(
                    color.g,
                    amount >> 1
                );
        }


        /*
         * --------------------------------------------------------
         * FINAL LEVEL CONTROL
         * --------------------------------------------------------
         */

        color.nscale8(
            225
        );


        leds[i] =
            color;
    }


    /*
     * One gentle spatial blend.
     */

    smoothField();
}


/* ================================================================
 *                         SETUP
 * ================================================================ */

void setup()
{
    /*
     * FastLED initialization.
     */

    FastLED.addLeds<
        WS2812B,
        LED_PIN,
        GRB
    >(
        leds,
        NUM_LEDS
    );


    /*
     * Global brightness.
     */

    FastLED.setBrightness(
        BRIGHTNESS
    );


    /*
     * Disable dithering.
     *
     * This is useful for this particular effect because the
     * animation itself already provides enough movement.
     */

    FastLED.setDither(
        0
    );


    /*
     * Clear strip.
     */

    FastLED.clear(
        true
    );


    /*
     * Deterministic starting point.
     *
     * Change these if you want a different starting pattern.
     */

    noiseTime =
        0x13579BDF;
}


/* ================================================================
 *                         MAIN LOOP
 * ================================================================ */

void loop()
{
    EVERY_N_MILLISECONDS(
        FRAME_MS
    )
    {
        /*
         * Move the large field.
         *
         * This is intentionally slow.
         *
         * The LED pattern should look like something moving
         * through the material rather than pixels scrolling.
         */

        noiseTime +=
            420;


        /*
         * Render.
         */

        renderFire();


        /*
         * Send the frame.
         */

        FastLED.show();
    }
}
